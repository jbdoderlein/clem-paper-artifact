% Questions work both inside and outside an enumerate environment.
% Inside enumerate they are numbered by the list; outside they have no number.
\newlength{\questionsep}\setlength{\questionsep}{0.9em}

\makeatletter
\newcommand{\@qstart}{%
  \ifnum\@enumdepth>0
    \item%                                  inside enumerate: list numbering
  \else
    \par\addvspace{\questionsep}\noindent%  standalone: no number
  \fi}
\newcommand{\question}[1]{\@qstart\textbf{#1}}

% Space kept between the end of the text and the rating boxes
\newlength{\ratinggap}\setlength{\ratinggap}{1em}

% \rating{...} bold statement; \rating*{...} plain statement
% Long statements wrap (ragged-right) inside the column to the left of the
% scale; the dotted leader and the boxes always sit at the end of the last line.
\NewDocumentCommand{\rating}{s m}{%
  \@qstart
  \begingroup
    \rightskip=\glueexpr\scalewidth+\ratinggap+0pt plus 1fil\relax
    \parfillskip=-\rightskip
    \IfBooleanTF{#1}{#2}{\textbf{#2}}%
    \nobreak\dotfill\nobreak
    \hbox to \dimexpr\scalewidth+\ratinggap\relax{\hfil\scale}%
    \par
  \endgroup}

% Scale labels above the rating boxes: right label on top, left label below.
% Usage: \ratingheader  or  \ratingheader[strongly disagree][strongly agree]
\NewDocumentCommand{\ratingheader}{O{novice} O{expert}}{%
  \ifnum\@enumdepth>0 \item[]\else\par\addvspace{\questionsep}\noindent\fi
  {\small\hfill #2 $\rightarrow$\\
   \null\hfill\makebox[\scalewidth][l]{$\leftarrow$ #1}\par}%
  \vspace{-0.4em}}
